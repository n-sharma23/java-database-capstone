# Step 1: log in as the patient. The response contains a JWT (shortened here).
$ curl -s -X POST http://localhost:8080/patient/login -H "Content-Type: application/json" -d '{"email":"john.smith@mail.example","password":"patient@123"}'
{"token": "eyJhbGciOiJIUzUxMiJ9.eyJ...Ut8muZ9u5Q"}

# Step 2: get the patient id from the token.
$ curl -s http://localhost:8080/patient/$TOKEN
{
    "patient": {
        "id": 1,
        "name": "John Smith",
        "email": "john.smith@mail.example",
        "phone": "8881112222",
        "address": "12 Oak Street, Springfield"
    }
}
